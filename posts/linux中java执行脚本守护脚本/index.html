#/bin/bash
while true;
do
count=`ps -ef | grep test.jar | grep -v grep|wc -l`
if [ ${count} -lt 1 ]; then
nohup java -jar xxx.jar >log.out 2>&1  &
else
echo “process is running”
fi
sleep 3
done
