开发系统的时候，我们避免不了要发送邮件，在php中使用phpmailer可以快速的实现邮件的发送

`请自行引用phpmailer模块，放下面这个文件和phpmailer同级 配置号里面的发送邮箱内容，执行php sendMail.php文件就可以测试发送邮件了
这个文件名称为sendMail.php
&lt;?php
use phpmailer\phpmailer\PHPMailer;
use phpmailer\phpmailer\Exception;
//引入项目
require './phpmailer/src/Exception.php';
require './phpmailer/src/PHPMailer.php';
require './phpmailer/src/SMTP.php';

//实例化PHPMail类
&lt;span class="katex math inline">mail = new PHPMailer(true);
try {
    //Server settings&lt;/span>mail->SMTPDebug = 2;
    &lt;span class="katex math inline">mail->isSMTP();&lt;/span>mail->Host = 'smtp.163.com';
    &lt;span class="katex math inline">mail->SMTPAuth = true;&lt;/span>mail->Username = 'hn_zhangdl@163.com';
    &lt;span class="katex math inline">mail->Password = 'xxxx';#跟上边一样的授权码&lt;/span>mail->SMTPSecure = 'ssl';
    &lt;span class="katex math inline">mail->Port = 994;&lt;/span>mail->CharSet='UTF-8';

    //发件人
    &lt;span class="katex math inline">mail->setFrom('hn_zhangdl@163.com', 'Mailer');

    //收件人。多收件人可设置多个addAddress&lt;/span>mail->addAddress('411437734@qq.com', 'zhangdl');//收件人邮箱地址，收件人姓名（选填）


    //发送附件
    //#&lt;span class="katex math inline">mail->addAttachment('/var/tmp/file.tar.gz'); // 添加附件
    //#&lt;/span>mail->addAttachment('/tmp/image.jpg', 'new.jpg'); // 设置附件以及附件名称

    //邮件内容
    &lt;span class="katex math inline">mail->isHTML(true); // 发送html格式邮件&lt;/span>mail->Subject = '标题'; //邮件标题
    &lt;span class="katex math inline">mail->Body    = '邮件测试内容 hello.';&lt;/span>mail->send();
    echo 'Message has been sent';

    echo 'Message could not be sent. Mailer Error: ',&lt;/span>mail->ErrorInfo;
}
